\usepackage{longtable}

\newcounter{tmp}
% Usage: \begin{usecase}{Title of the use case}{uc-01} ... \end{usecase}
% The second argument names the table: it is labelled tab:<id>, so the text can cite it with Table~\ref{tab:uc-01}.
\newcommand\tabularhead[2]{
% \begin{table}[h]
  \begin{longtable}{|p{0.3\linewidth}|p{0.55\linewidth}|}
  \caption{Use case: #1}\label{tab:#2} \\
    \hline
    \textbf{Use-case} & {\textbf{#1}} \\
    \hline}

  \newcommand\addrow[2]{#1 & #2 \\ \hline}

  \newcommand\addmulrow[2]{ #1 
     &  \begin{enumerate} #2  \setcounter{tmp}{\value{enumi}}\end{enumerate}  \\ \hline}
     
  \newcommand\addbreakmulrow[1]{ 
     &  \begin{enumerate}\setcounter{enumi}{\value{tmp}} #1  \end{enumerate}  \\ \hline}
  
 
  \newenvironment{usecase}{\begingroup\setstretch{1.15}\tabularhead}
{\end{longtable}\endgroup}
% \end{table}}
