% A UML actor: stick figure with its name below. Usage: \umlactor{name}{(x,y)}{Label}
\newcommand{\umlactor}[3]{%
    \begin{scope}[shift={#2}, thick]
        \draw (0,0.55) circle (0.13);
        \draw (0,0.42) -- (0,0.05);
        \draw (-0.22,0.30) -- (0.22,0.30);
        \draw (0,0.05) -- (-0.18,-0.25);
        \draw (0,0.05) -- (0.18,-0.25);
        \node[coordinate] (#1) at (0,0.2) {};
        \node[align=center, font=\scriptsize, anchor=north] (#1-label) at (0,-0.28) {#3};
    \end{scope}}
% Stereotype line above a name, e.g. \stereo{component}
\newcommand{\stereo}[1]{{\fontencoding{T1}\selectfont\guillemotleft}#1{\fontencoding{T1}\selectfont\guillemotright}}
