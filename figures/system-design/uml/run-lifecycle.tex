% UML state machine diagram of a run (Section 5.5.1).
\begin{tikzpicture}[uml]
    \node[initial] (init) at (0,0) {};
    \node[state] (queued) at (2.2,0) {queued};
    \node[state] (running) at (5.4,0) {running};
    \node[state] (interrupted) at (5.4,2.2) {interrupted};
    \node[state] (succeeded) at (9.4,0.7) {succeeded};
    \node[state] (failed) at (9.4,-0.7) {failed};
    \node[state] (cancelled) at (9.4,2.2) {cancelled};
    \node[final] (fin) at (11.8,0.7) {};

    \draw[trans] (init) -- node[guard, above] {message posted} (queued);
    \draw[trans] (queued) -- node[guard, above] {started} (running);
    \draw[trans] (running.north west) to[bend left=20] node[guard, left] {checkpoint reached} (interrupted.south west);
    \draw[trans] (interrupted.south east) to[bend left=20] node[guard, right] {decision received} (running.north east);
    \draw[trans] (running) -- node[guard, above, sloped] {end of workflow} (succeeded.west);
    \draw[trans] (running) -- node[guard, below, sloped] {step failed / server\_restarted} (failed.west);
    \draw[trans] (running) -- node[guard, above, sloped, pos=0.8] {cancel} (cancelled.south west);
    \draw[trans] (queued.north) |- node[guard, pos=0.75, above] {cancel} ($(cancelled.north)+(0,0.6)$) -- (cancelled.north);
    \draw[trans] (interrupted.east) -- node[guard, above] {cancel} (cancelled.west);
    \draw[trans] (succeeded.east) -- (fin);
    \draw[trans] (failed.east) -| (fin);
    \draw[trans] (cancelled.east) -| (fin);
\end{tikzpicture}
